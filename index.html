import kagglehub
import pandas as pd
import os
import plotly.express as px
import plotly.graph_objects as go
import folium

# Download and load official F1 dataset dynamically
dataset_path = kagglehub.dataset_download("rohanrao/formula-1-world-championship-1950-2020")
df_races = pd.read_csv(os.path.join(dataset_path, "races.csv"))
df_constructor_standings = pd.read_csv(os.path.join(dataset_path, "constructor_standings.csv"))

# Filter for 2022 season and merge circuits for geospatial mapping
df_races = df_races[df_races['year'] == 2022]
df_races = df_races.merge(pd.read_csv(os.path.join(dataset_path, "circuits.csv"))[['circuitId', 'lat', 'lng']], on='circuitId', how='left')

# Generate interactive Folium Map
circuit_map = folium.Map(location=[df_races['lat'].mean(), df_races['lng'].mean()], zoom_start=3)
for idx, row in df_races.iterrows():
    folium.Marker(location=[row['lat'], row['lng']], popup=row['name']).add_to(circuit_map)
